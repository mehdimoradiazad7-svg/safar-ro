create table if not exists public.profiles (
  id uuid primary key references auth.users(id) on delete cascade,
  display_name text not null default 'کاربر',
  phone_e164 text unique,
  role public.app_role not null default 'passenger',
  created_at timestamptz not null default now(),
  updated_at timestamptz not null default now()
);

create table if not exists public.driver_profiles (
  user_id uuid primary key
    references public.profiles(id) on delete cascade,
  verification_status text not null default 'pending'
    check (verification_status in (
      'pending', 'approved', 'rejected', 'suspended'
    )),
  vehicle_type public.service_type not null default 'car',
  vehicle_make text,
  vehicle_model text,
  plate_number text,
  is_online boolean not null default false,
  created_at timestamptz not null default now()
);

create table if not exists public.rides (
  id uuid primary key default gen_random_uuid(),
  passenger_id uuid not null references public.profiles(id),
  driver_id uuid references public.profiles(id),
  service_type public.service_type not null default 'car',
  pickup_label text not null,
  pickup_lat double precision,
  pickup_lng double precision,
  destination_label text not null,
  destination_lat double precision,
  destination_lng double precision,
  estimated_fare_irr bigint check (estimated_fare_irr >= 0),
  final_fare_irr bigint check (final_fare_irr >= 0),
  status public.ride_status not null default 'requested',
  requested_at timestamptz not null default now(),
  accepted_at timestamptz,
  arrived_at timestamptz,
  wait_timer_started_at timestamptz,
  cancelled_at timestamptz,
  cancellation_reason public.cancellation_reason,
  completed_at timestamptz,
  created_at timestamptz not null default now(),
  constraint ride_coords_pair check (
    ((pickup_lat is null) = (pickup_lng is null))
    and ((destination_lat is null) = (destination_lng is null))
  ),
  constraint ride_driver_not_passenger check (
    driver_id is null or driver_id <> passenger_id
  )
);SQL
create table if not exists public.promo_codes (
  id uuid primary key default gen_random_uuid(),
  code text not null unique,
  discount_percent numeric(5,2)
    check (discount_percent > 0 and discount_percent <= 100),
  discount_amount_irr bigint check (discount_amount_irr >= 0),
  starts_at timestamptz not null default now(),
  ends_at timestamptz not null,
  max_uses integer check (max_uses is null or max_uses > 0),
  max_uses_per_user integer not null default 1
    check (max_uses_per_user > 0),
  is_active boolean not null default false,
  created_at timestamptz not null default now(),
  constraint promo_valid_dates check (ends_at > starts_at),
  constraint promo_one_discount_type check (
    (discount_percent is not null)::int
    + (discount_amount_irr is not null)::int = 1
  )
);

create table if not exists public.promo_redemptions (
  id uuid primary key default gen_random_uuid(),
  promo_code_id uuid not null references public.promo_codes(id),
  user_id uuid not null references public.profiles(id),
  ride_id uuid not null unique references public.rides(id),
  discount_amount_irr bigint not null
    check (discount_amount_irr >= 0),
  redeemed_at timestamptz not null default now()
);

create table if not exists public.driver_monthly_rewards (
  id uuid primary key default gen_random_uuid(),
  driver_id uuid not null references public.profiles(id),
  period_start date not null,
  completed_eligible_rides integer not null default 0
    check (completed_eligible_rides >= 0),
  fuel_liters numeric(6,2) not null default 0
    check (fuel_liters >= 0),
  shopping_credit_irr bigint not null default 0
    check (shopping_credit_irr >= 0),
  status text not null default 'pending'
    check (status in ('pending','approved','fulfilled','rejected')),
  reviewed_by uuid references public.profiles(id),
  reviewed_at timestamptz,
  created_at timestamptz not null default now(),
  unique (driver_id, period_start)
);

create table if not exists public.emergency_alerts (
  id uuid primary key default gen_random_uuid(),
  ride_id uuid references public.rides(id),
  triggered_by uuid not null references public.profiles(id),
  latitude double precision,
  longitude double precision,
  alert_type text not null default 'safety_concern',
  status text not null default 'open'
    check (status in ('open','acknowledged','resolved','false_alarm')),
  created_at timestamptz not null default now(),
  resolved_at timestamptz,
  resolved_by uuid references public.profiles(id)
);SQL
create table if not exists public.ride_events (
  id uuid primary key default gen_random_uuid(),

create or replace function public.handle_new_user()
returns trigger
language plpgsql
security definer
set search_path = ''
as $$
begin
  insert into public.profiles (id, display_name)
  values (
    new.id,
    coalesce(
      nullif(new.raw_user_meta_data ->> 'display_name', ''),
      'کاربر'
    )
  )
  on conflict (id) do nothing;

  return new;
end;
$$;

drop trigger if exists on_auth_user_created on auth.users;

create trigger on_auth_user_created
after insert on auth.users
for each row
execute function public.handle_new_user();SQL
create or replace function public.current_app_role()
returns public.app_role
language sql
stable
security definer
set search_path = ''
as $$
  select p.role
  from public.profiles p
  where p.id = auth.uid()
$$;

create or replace function public.is_admin()
returns boolean
language sql
stable
security definer
set search_path = ''
as $$
  select coalesce(
    public.current_app_role() = 'admin'::public.app_role,
    false
  )
$$;

revoke all on function public.current_app_role()
from public, anon;

revoke all on function public.is_admin()
from public, anon;

grant execute on function public.current_app_role()
to authenticated;

grant execute on function public.is_admin()
to authenticated;SQL

create or replace function public.request_ride(
  p_service_type public.service_type,
  p_pickup_label text,
  p_pickup_lat double precision,
  p_pickup_lng double precision,
  p_destination_label text,
  p_destination_lat double precision,
  p_destination_lng double precision,
  p_estimated_fare_irr bigint default null
)
returns uuid
language plpgsql
security definer
set search_path = ''
as $$
declare
  v_user uuid := auth.uid();
  v_ride_id uuid;
begin
  if v_user is null then
    raise exception 'Authentication required';
  end if;

  if nullif(trim(p_pickup_label), '') is null
     or nullif(trim(p_destination_label), '') is null then
    raise exception 'Pickup and destination are required';
  end if;

  if (p_pickup_lat is null) <> (p_pickup_lng is null)
     or (p_destination_lat is null) <> (p_destination_lng is null) then
    raise exception 'Coordinates must be supplied in pairs';
  end if;

  if p_estimated_fare_irr is not null
     and p_estimated_fare_irr < 0 then
    raise exception 'Invalid estimated fare';
  end if;

  insert into public.rides (
    passenger_id, service_type,
    pickup_label, pickup_lat, pickup_lng,
    destination_label, destination_lat, destination_lng,
    estimated_fare_irr
  )
  values (
    v_user, p_service_type,
    trim(p_pickup_label), p_pickup_lat, p_pickup_lng,
    trim(p_destination_label), p_destination_lat, p_destination_lng,
    p_estimated_fare_irr
  )
  returning id into v_ride_id;

  insert into public.ride_events (ride_id, actor_id, event_type)
  values (v_ride_id, v_user, 'ride_requested');

  return v_ride_id;
end;
$$;

revoke all on function public.request_ride(
  public.service_type, text, double precision, double precision,
  text, double precision, double precision, bigint
) from public, anon;

grant execute on function public.request_ride(
  public.service_type, text, double precision, double precision,
  text, double precision, double precision, bigint
) to authenticated;SQL
create or replace function public.accept_ride(p_ride_id uuid)
returns uuid
language plpgsql
security definer
set search_path = ''
as $$
declare
  v_user uuid := auth.uid();
  v_ride_id uuid;
begin
  if v_user is null then
    raise exception 'Authentication required';
  end if;

  if not exists (
    select 1
    from public.driver_profiles d
    join public.profiles p on p.id = d.user_id
    where d.user_id = v_user
      and d.verification_status = 'approved'
      and d.is_online = true
      and p.role = 'driver'
  ) then
    raise exception 'Approved online driver account required';
  end if;

  update public.rides
  set driver_id = v_user,
      status = 'accepted',
      accepted_at = now()
  where id = p_ride_id
    and status = 'requested'
    and passenger_id <> v_user
  returning id into v_ride_id;

  if v_ride_id is null then
    raise exception 'Ride is unavailable';
  end if;

  insert into public.ride_events (ride_id, actor_id, event_type)
  values (v_ride_id, v_user, 'ride_accepted');

  return v_ride_id;
end;
$$;

revoke all on function public.accept_ride(uuid) from public, anon;
grant execute on function public.accept_ride(uuid) to authenticated;SQL
create or replace function public.update_ride_status(
  p_ride_id uuid,
  p_new_status public.ride_status,
  p_final_fare_irr bigint default null,
  p_cancellation_reason public.cancellation_reason default null
)
returns uuid
language plpgsql
security definer
set search_path = ''
as $$
declare
  v_user uuid := auth.uid();
  v_ride public.rides%rowtype;
begin
  if v_user is null then
    raise exception 'Authentication required';
  end if;

  select * into v_ride
  from public.rides
  where id = p_ride_id
  for update;

  if not found then
    raise exception 'Ride not found';
  end if;

  if p_new_status = 'driver_arrived' then
    if v_ride.driver_id is distinct from v_user
       or v_ride.status <> 'accepted' then
      raise exception 'Invalid ride transition';
    end if;

    update public.rides
    set status = 'driver_arrived',
        arrived_at = now(),
        wait_timer_started_at = now()
    where id = p_ride_id;

  elsif p_new_status = 'in_progress' then
    if v_ride.driver_id is distinct from v_user
       or v_ride.status <> 'driver_arrived' then
      raise exception 'Invalid ride transition';
    end if;

    update public.rides
    set status = 'in_progress'
    where id = p_ride_id;

  elsif p_new_status = 'completed' then
    if v_ride.driver_id is distinct from v_user
       or v_ride.status <> 'in_progress' then
      raise exception 'Invalid ride transition';
    end if;

    if p_final_fare_irr is null or p_final_fare_irr < 0 then
      raise exception 'A valid final fare is required';
    end if;

    update public.rides
    set status = 'completed',
        final_fare_irr = p_final_fare_irr,
        completed_at = now()
    where id = p_ride_id;

  elsif p_new_status = 'cancelled' then
    if v_ride.status in ('completed', 'cancelled') then
      raise exception 'Ride cannot be cancelled';
    end if;

    if v_ride.passenger_id <> v_user
       and v_ride.driver_id is distinct from v_user then
      raise exception 'Only ride participants can cancel';
    end if;

    update public.rides
    set status = 'cancelled',
        cancelled_at = now(),
        cancellation_reason = coalesce(p_cancellation_reason, 'other')
    where id = p_ride_id;

  else
    raise exception 'Unsupported ride transition';
  end if;

  insert into public.ride_events (ride_id, actor_id, event_type)
  values (p_ride_id, v_user, 'status_' || p_new_status::text);

  return p_ride_id;
end;
$$;

revoke all on function public.update_ride_status(
  uuid, public.ride_status, bigint, public.cancellation_reason
) from public, anon;

grant execute on function public.update_ride_status(
  uuid, public.ride_status, bigint, public.cancellation_reason
) to authenticated;SQL
create or replace function public.approve_driver(
  p_driver_id uuid,
  p_decision text
)
returns void
language plpgsql
security definer
set search_path = ''
as $$
begin
  if auth.uid() is null or not public.is_admin() then
    raise exception 'Administrator access required';
  end if;

  if p_decision not in ('approved', 'rejected', 'suspended') then
    raise exception 'Invalid decision';
  end if;

  update public.driver_profiles
  set verification_status = p_decision,
      is_online = false
  where user_id = p_driver_id;

  if not found then
    raise exception 'Driver profile not found';
  end if;

  if p_decision = 'approved' then
    update public.profiles
    set role = 'driver'
    where id = p_driver_id;
  end if;
end;
$$;

revoke all on function public.approve_driver(uuid, text)
from public, anon;

grant execute on function public.approve_driver(uuid, text)
to authenticated;
to authenticated;SQL
